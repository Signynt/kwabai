 # Track installed ABI headers as configure inputs, not just project sources.
 get_target_property(kwabai_kwin_includes KWin::kwin INTERFACE_INCLUDE_DIRECTORIES)
 find_path(KWABAI_KWIN_HEADER_DIR config-kwin.h HINTS ${kwabai_kwin_includes} REQUIRED NO_DEFAULT_PATH)

 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${kwabai_kwin_headers})
 file(STRINGS "${KWABAI_KWIN_HEADER_DIR}/config-kwin.h" kwabai_kwin_version_line REGEX "^#define KWIN_PLUGIN_VERSION_STRING ")
 string(REGEX REPLACE "^#define KWIN_PLUGIN_VERSION_STRING \"([^\"]+)\".*" "\\1" KWABAI_KWIN_VERSION "${kwabai_kwin_version_line}")
 if(NOT KWABAI_KWIN_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Cannot determine the installed KWin plugin ABI version")
 endif()
 # A unique module path avoids resident Qt plugin code being reused after reload.
 set(kwabai_effect_sources src/probe/build-version.h.in src/compat/color.h src/probe/drop_overlay.h src/probe/probe.cpp src/probe/bridge.cpp src/probe/bridge.h src/probe/metadata.json src/bridge/events.cpp src/bridge/events.h src/bridge/geometry.cpp src/bridge/geometry.h src/bridge/focus.cpp src/bridge/focus.h src/bridge/desktops.cpp src/bridge/desktops.h src/bridge/layers.cpp src/bridge/layers.h src/bridge/scratchpads.cpp src/bridge/scratchpads.h)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${kwabai_effect_sources})
 set(kwabai_effect_fingerprint "${CMAKE_CXX_COMPILER_ID};${CMAKE_CXX_COMPILER_VERSION};${Qt6_VERSION};${KWABAI_KWIN_VERSION}")
 foreach(header IN LISTS kwabai_kwin_headers)
  file(SHA256 "${header}" header_hash)
  string(APPEND kwabai_effect_fingerprint ";${header_hash}")
 endforeach()
 foreach(source IN LISTS kwabai_effect_sources)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND kwabai_effect_fingerprint ";${source_hash}")
 endforeach()
 string(SHA256 kwabai_effect_hash "${kwabai_effect_fingerprint}")
 string(SUBSTRING "${kwabai_effect_hash}" 0 12 kwabai_effect_hash)
 string(REPLACE "." "_" kwabai_kwin_id "${KWABAI_KWIN_VERSION}")
 set(KWABAI_EFFECT_ID "kwabai_probe_kwin_${kwabai_kwin_id}_${kwabai_effect_hash}")
 configure_file(src/probe/metadata.json "${CMAKE_BINARY_DIR}/kwabai-metadata.json" @ONLY)
 configure_file(src/probe/build-version.h.in "${CMAKE_BINARY_DIR}/kwabai-build-version.h" @ONLY)
 configure_file(scripts/native-build.json.in "${CMAKE_BINARY_DIR}/kwabai-native-build.json" @ONLY)
 target_include_directories(kwabai_probe PRIVATE "${CMAKE_BINARY_DIR}")
 set_property(TARGET kwabai_probe APPEND PROPERTY AUTOGEN_TARGET_DEPENDS "${CMAKE_BINARY_DIR}/kwabai-metadata.json" "${CMAKE_BINARY_DIR}/kwabai-build-version.h" ${kwabai_kwin_headers})
 add_executable(kwabai_plugin_check src/probe/plugin_check.cpp)
 set_target_properties(kwabai_plugin_check PROPERTIES OUTPUT_NAME "kwabai-plugin-check")
 target_link_libraries(kwabai_plugin_check PRIVATE KWin::kwin KF6::CoreAddons)
 add_dependencies(kwabai_plugin_check kwabai_probe)
 install(TARGETS kwabai_plugin_check RUNTIME DESTINATION bin)
 file(WRITE "${CMAKE_BINARY_DIR}/kwabai-effect-id" "${KWABAI_EFFECT_ID}\n")
 set_target_properties(kwabai_probe PROPERTIES PREFIX "" OUTPUT_NAME "${KWABAI_EFFECT_ID}")
 target_link_libraries(kwabai_probe PRIVATE KWin::kwin Qt6::DBus kwabai_event_protocol Plasma::Activities)
 target_compile_options(kwabai_probe PRIVATE -Wall -Wextra -Wpedantic)
 # Reject private/header-only KWin APIs absent from the exported plugin ABI.
 target_link_options(kwabai_probe PRIVATE "LINKER:-z,defs")
endif()
